# 接入指南

三个模块都是库,需要宿主进程(FastAPI 应用或任意 asyncio 程序)在启动时完成接线。本文给出**可直接抄写的最小接线示例**,mock 部分替换成你自己的实现即可。

## 0. 前置:LLM 调用函数

所有模块消费 LLM 的方式都是"注入一个异步函数"。先准备两档:

```python
from openai import AsyncOpenAI
import os

_pro = AsyncOpenAI(api_key=os.getenv("DEEPSEEK_API_KEY"),
                   base_url="https://api.deepseek.com")
_flash_model = os.getenv("DEEPSEEK_FLASH_MODEL", "deepseek-v4-flash")
_pro_model = os.getenv("DEEPSEEK_MODEL", "deepseek-v4-pro")


async def call_llm_flash_for_text(prompt: str, max_tokens: int = 1024) -> str:
    """轻模型:内部摘要/意图判断。入参纯文本,返回纯文本。"""
    resp = await _pro.chat.completions.create(
        model=_flash_model,
        messages=[{"role": "user", "content": prompt}],
        max_tokens=max_tokens, temperature=0.3)
    return resp.choices[0].message.content or ""


async def call_llm_api(messages: list, provider: str = "deepseek") -> dict:
    """主模型:面向用户的生成。入参 messages 数组,返回 {content, reasoning}。"""
    resp = await _pro.chat.completions.create(
        model=_pro_model, messages=messages, max_tokens=2048)
    msg = resp.choices[0].message
    return {"content": msg.content or "",
            "reasoning": getattr(msg, "reasoning_content", None)}
```

## 1. 硅基感知(silicon_perception)

```python
from silicon_perception import init_sentinel


async def on_sentinel_push(message: str, event: dict):
    """哨兵决定推送时回调——把 message 发到你的前端/IM/通知渠道。"""
    print(f"[哨兵] {message}")


async def perception_push_pro_llm(summary: str) -> dict:
    """哨兵感知到值得开口的变化时,用主模型生成一句自然语言。"""
    system_ctx = f"感知系统注意到:{summary}"
    push_instruction = (
        "你刚才通过传感器感知到了用户的一些变化。"
        "你决定主动和TA说句话——这是你自己的选择,不是任何人要求的。\n\n"
        "一切正常就轻松调侃一句,有值得关心的就温和地问一下。"
        "别说教、别重复数据、一句就够了。"
    )
    return await call_llm_api([
        {"role": "system", "content": system_ctx + "\n\n" + push_instruction},
    ])


sentinel = init_sentinel(
    on_push=on_sentinel_push,
    tick_interval=60,                    # 监控循环间隔(秒)
    enable_llm_polish=False,
    call_llm_func=call_llm_flash_for_text,   # Flash:每 tick 感知摘要
    call_pro_llm=perception_push_pro_llm,    # Pro:推送文案生成
)
await sentinel.start()      # 启动 60s tick 循环
# 关闭时: await sentinel.stop()
```

数据源可用性:PC 键鼠/屏幕开箱即用;心率需 `HR_DEVICE_MAC` + bleak;GPS/步数/屏幕时间需手机中继(见 §4);天气需 `WEATHER_CITY`。缺什么降级什么,不会崩。

## 2. 漫想(wander_manager)

```python
from wander_manager import init_wander_manager


async def push_wander_message(payload: dict):
    """漫想消息推送回调——payload 含 content/event_type 等,发给你的前端。"""
    print(f"[漫想] {payload.get('content')}")


def on_status_change(payload: dict):
    """用户状态变化回调(同步),用于前端状态面板刷新。可以是空实现。"""
    pass


async def get_recent_user_messages(limit: int = 10) -> list:
    """返回最近的用户消息文本列表,供漫想取材。接你自己的会话存储。"""
    return []


wander = init_wander_manager(
    idle_threshold=10 * 60,       # 用户空闲多久后进入漫想(秒)
    event_interval=15 * 60,       # 漫想事件最小间隔(秒)
    push_threshold=0.6,           # 推送阈值(0-1,越高越沉默)
    call_llm_func=call_llm_api,           # 生成/创意任务

    on_push_to_user=push_wander_message,
    on_status_change=on_status_change,
    web_search_func=None,                 # 可选:看新闻事件需要
    persistence_enabled=True,
    get_recent_user_messages_func=get_recent_user_messages,
    get_memories_func=None,               # 可选:记忆抓取,协议见 记忆接口协议.md
)

# 用户每次发消息时通知漫想(重置空闲计时):

```

## 3. 工具调用(behavior_scheduler)

```python
from behavior_scheduler import init_global_scheduler

scheduler = init_global_scheduler(
    call_llm_func=call_llm_api,          # 兼容回退路径
    # 视觉模型(截屏/摄像头分析;不用视觉工具可传空)
    glm_api_key=os.getenv("GLM_API_KEY", ""),
    glm_api_url=os.getenv("GLM_API_URL", ""),
    glm_model=os.getenv("GLM_MODEL", "glm-4v-flash"),
    use_agent_tools=True,
    # Flash 适配器(后续轮次/意图兜底提取)
    llm_provider="deepseek",
    llm_api_key=os.getenv("DEEPSEEK_API_KEY"),
    llm_base_url="https://api.deepseek.com",
    llm_model=os.getenv("DEEPSEEK_FLASH_MODEL"),
    # Pro 适配器(首轮聊天+工具决策,建议开 thinking)
    pro_api_key=os.getenv("DEEPSEEK_API_KEY"),
    pro_model=os.getenv("DEEPSEEK_MODEL"),
    pro_base_url="https://api.deepseek.com",
)

# 处理一条用户消息(persona_prompt = 你的 AI 人设,名字/性格都写在这里):
result = await scheduler.process_message(
    user_message="帮我搜一下明天的天气",
    conversation_history=[],
    persona_prompt="你是用户的AI伙伴,说话简洁。",
)
print(result.final_response)
```

三通道检测:Pre-tool 关键词直接命中(零 LLM 成本)→ Pro 输出 TOOL_CALL(快速路径)→ Flash 结构化提取(兜底)。注册自定义工具:继承 `base_tool.BaseTool`,在 `tools.py` 的 `register_all_tools()` 注册,并同步三处——`intent_parser.TOOL_KEYWORD_MAP` 关键词、scheduler 的 `task_instruction` 工具描述、(若参数每次可变)设置 `single_use = True`。

## 4. 全局状态接线(mirrow_core.shared_state)

宿主进程负责在恰当时机写入,模块只读:

```python
from mirrow_core import shared_state

# 前端心跳(每 3-5s 上报一次,用于"用户是否在看界面"):
shared_state.set_mirrow_page_visible(True)
shared_state.set_mirrow_page_last_seen(time.time())

# 手机端工具中继(可选,无手机端跳过):
# 签名: async def relay(request_id: str, tool: str, params: dict) -> dict
shared_state.set_mobile_relay_callback(my_relay)
shared_state.incr_mobile_ws()   # 手机 WS 连上时 / decr_mobile_ws() 断开时

# 勿扰开关(压制哨兵/漫想所有主动推送):
shared_state.set_dnd_enabled(True)
```

## 5. 称呼配置(mirrow_core.persona)

展示文案(如行为状态"用户在用电脑")的称呼从这里取:

```
PERSONA_USER_NAME=你的名字     # .env,默认"用户"
PERSONA_AI_NAME=你的AI的名字   # 默认"AI"
```

AI 的完整人设(性格/说话风格/自称)不在这里——通过 `process_message` 的 `persona_prompt` 参数和各推送回调里你自己写的 prompt 注入。

## 6. 可选:接线上下文组装(context_builder,批次2)

如果你想让 AI 拥有"记得昨天聊过什么/记得三个月前的旅行计划/知道今天是什么日子"的能力——这是上下文模块做的事:

```python
from context_builder import ContextBuilder
from memory_provider import SQLiteMemoryProvider

builder = ContextBuilder()

# 你自己的记忆存储(不接也行)
provider = SQLiteMemoryProvider("data/memories.db")

# 每轮对话前构建上下文
ctx = builder.build(
    "FULL_CHAT",
    persona="你是用户的AI伴侣,说话温柔。",
    messages=conversation_history,          # 最近 N 条对话
    memory_provider=provider,               # ← 不传=记忆段自动为空
    # 以下全部可选,不传=对应段降级为空
    user_status="awake",                    # 用户当前状态
    user_name="用户",                       # 用户称呼
    night_mode=False,                       # 是否深夜模式
    wander_mode=False,                      # 是否漫想模式
)

# ctx.system_content → 给 LLM 的 system prompt
# ctx.formatted_messages → 对话消息(角色前缀已加好)
# ctx.recipe_stats → 各段状态{内存中可以看到哪些段生效了}
```

**不接任何数据源时**,Builder 照样输出一段只含 A 层(你是谁)+ B 层(时间)+ E 层(当前对话)的 system prompt——AI 的核心人格一致性不依赖任何外部数据。

完整设计哲学和注意力分层理论见「上下文组装方法论」。

## 7. 常见降级行为速查

| 缺失项 | 行为 |
|---|---|
| `get_memories_func` 未注入 | 漫想"翻记忆"事件返回空,其他事件正常 |
| 手机中继未接 | GPS/步数/屏幕时间/手机截图采集静默跳过 |
| `HR_DEVICE_MAC` 未配 | 心率采集跳过,相关规则不触发 |
| GLM key 未配 | 截屏/摄像头工具返回"视觉分析不可用" |
| `WEATHER_CITY` 未配 | 天气采集跳过 |
| bleak/mss/cv2 未安装 | 对应数据源 import 降级,其余正常 |
